import pandas as pd import numpy as np # ========================================== # 1. データの読み込み # ========================================== # 実際の運用では以下のようにCSVファイルを読み込んでください。 # df = pd.read_csv('sleeping_det_data.csv') # 【テスト用】5分間隔の居眠り検知ダミーデータを生成（2026年5月28日〜29日） date_rng = pd.date_range(start='2026-05-28 08:30:00', end='2026-05-29 15:30:00', freq='5min') df = pd.DataFrame(date_rng, columns=['timestamp']) # ダミーの居眠りフラグ（0: 起きている、1: 居眠り検知）を生成 df['sleeping_detected'] = np.random.choice([0, 1], size=(len(date_rng)), p=[0.85, 0.15]) # 瞬き（まばたき）の間隔や、頭の傾き角度（度）などのセンサー値の想定 df['blink_interval'] = np.random.uniform(1.0, 6.0, size=(len(date_rng))) df['head_tilt'] = np.random.uniform(0.0, 45.0, size=(len(date_rng))) # ========================================== # 2. データの前処理 # ========================================== # 日時列をインデックスに設定 df['timestamp'] = pd.to_datetime(df['timestamp']) df.set_index('timestamp', inplace=True) # ========================================== # 3. 1日ごとの集計（リサンプリング） # ========================================== # 'D'（1日ごと）でグループ化します # 居眠りフラグは「合計(sum)」することで、1日の延べ居眠り検知回数（×5分）になります daily_summary = df.resample('D').agg({ 'sleeping_detected': 'sum', # 1日の総居眠り検知（回） 'blink_interval': 'mean', # 平均まばたき間隔 'head_tilt': 'max' # 最大の頭の傾き }) # ========================================== # 4. 結果の整形と出力 # ========================================== daily_summary.columns = ['居眠り検知回数（5分単位）', '平均まばたき間隔(秒)', '最大頭部傾き(度)'] daily_summary = daily_summary.round(1) print("【1日ごとの居眠りセンサー集計結果】") print(daily_summary) # CSVとして書き出す場合 # daily_summary.to_csv('daily_sleeping_summary.csv', encoding='utf-8-sig')
